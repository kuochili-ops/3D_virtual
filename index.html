import * as THREE from 'three'; class WordMarquee { constructor(scene, text, startPos, turnX) { this.scene = scene; this.characters = []; this.text = text; this.startPos = startPos; // {x, y, z} this.turnX = turnX; // 轉彎點的 X 座標 this.spacing = 2.5; // 字與字之間的間距 this.init(); } // 建立單個字的 3D 方塊 createCharMesh(char) { const canvas = document.createElement('canvas'); canvas.width = 256; canvas.height = 256; const ctx = canvas.getContext('2d'); // 繪製背景與文字 (保證中文顯示) ctx.fillStyle = '#000000'; ctx.fillRect(0, 0, 256, 256); ctx.fillStyle = '#00ffff'; ctx.font = 'bold 180px "Microsoft JhengHei"'; ctx.textAlign = 'center'; ctx.textBaseline = 'middle'; ctx.fillText(char, 128, 128); const texture = new THREE.CanvasTexture(canvas); const materials = [ new THREE.MeshPhongMaterial({ color: 0x0088ff }), // 側面 new THREE.MeshPhongMaterial({ color: 0x0088ff }), // 側面 new THREE.MeshPhongMaterial({ color: 0x0088ff }), // 頂面 new THREE.MeshPhongMaterial({ color: 0x0088ff }), // 底面 new THREE.MeshPhongMaterial({ map: texture }), // 正面 (顯示字) new THREE.MeshPhongMaterial({ color: 0x004488 }) // 背面 ]; const geometry = new THREE.BoxGeometry(2, 2, 0.8); // 0.8 就是厚度 const mesh = new THREE.Mesh(geometry, materials); return mesh; } init() { for (let i = 0; i < this.text.length; i++) { const charMesh = this.createCharMesh(this.text[i]); // 設定每個字的初始延遲位置 charMesh.position.set(this.startPos.x + (i * this.spacing), this.startPos.y, this.startPos.z); this.characters.push(charMesh); this.scene.add(charMesh); } } update() { this.characters.forEach((char) => { // 階段 1：向左前行進 if (char.position.x > this.turnX) { char.position.x -= 0.05; } // 階段 2：到達轉角右轉 (進入門) else { if (char.rotation.y > -Math.PI / 2) { char.rotation.y -= 0.1; // 平滑轉向動畫 } char.position.z -= 0.08; // 往深處走 } // 重置循環：當字完全消失後回到起點 if (char.position.z < -10) { char.position.set(this.startPos.x, this.startPos.y, this.startPos.z); char.rotation.y = 0; } }); } }
